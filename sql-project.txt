DROP DATABASE IF EXISTS banking_management;
CREATE DATABASE banking_management;
USE banking_management;

CREATE TABLE branches (
    branch_id INT AUTO_INCREMENT PRIMARY KEY,
    branch_name VARCHAR(100) NOT NULL,
    ifsc_code VARCHAR(11) NOT NULL UNIQUE,
    city VARCHAR(80) NOT NULL
);

CREATE TABLE customers (
    customer_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(120) NOT NULL,
    phone VARCHAR(20) UNIQUE,
    email VARCHAR(120) UNIQUE,
    address VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE employees (
    employee_id INT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL,
    employee_name VARCHAR(120) NOT NULL,
    job_title VARCHAR(60) NOT NULL,
    phone VARCHAR(20),
    FOREIGN KEY (branch_id) REFERENCES branches(branch_id)
);

CREATE TABLE accounts (
    account_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL,
    branch_id INT NOT NULL,
    account_number VARCHAR(20) NOT NULL UNIQUE,
    account_type ENUM('Savings','Current') NOT NULL,
    balance DECIMAL(14,2) NOT NULL DEFAULT 0.00 CHECK (balance >= 0),
    status ENUM('Active','Frozen','Closed') NOT NULL DEFAULT 'Active',
    opened_on DATE NOT NULL,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
    FOREIGN KEY (branch_id) REFERENCES branches(branch_id)
);

CREATE TABLE transactions (
    transaction_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    account_id BIGINT NOT NULL,
    transaction_type ENUM('Deposit','Withdrawal','Transfer In','Transfer Out') NOT NULL,
    amount DECIMAL(14,2) NOT NULL CHECK (amount > 0),
    transaction_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reference_note VARCHAR(255),
    FOREIGN KEY (account_id) REFERENCES accounts(account_id),
    INDEX idx_transaction_time (transaction_time),
    INDEX idx_transaction_account (account_id)
);

*JOINS& CASE OPERATOR,CLAUSES
-----------------------------
query 3: total deposits and withdrawals recorded by account?
=======
SELECT a.account_number,
       SUM(CASE WHEN t.transaction_type = 'Deposit' THEN t.amount ELSE 0 END) AS total_deposits,
       SUM(CASE WHEN t.transaction_type = 'Withdrawal' THEN t.amount ELSE 0 END) AS total_withdrawals
FROM accounts a
LEFT JOIN transactions t ON t.account_id = a.account_id
GROUP BY a.account_id, a.account_number
ORDER BY a.account_number;

query 4: transaction history for one account?
=======

SELECT t.transaction_id, a.account_number, t.transaction_type,
       t.amount, t.transaction_time, t.reference_note
FROM transactions t
JOIN accounts a ON a.account_id = t.account_id
WHERE a.account_number = '100000000001'
ORDER BY t.transaction_time DESC;

CREATE TRIGGER trg_check_balance
BEFORE INSERT ON accounts
FOR EACH ROW
BEGIN
    IF NEW.balance < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Balance cannot be negative';
    END IF;
END //
